cmake_minimum_required(VERSION 3.16)
project(PRISMConsumer LANGUAGES C)
set(PRISM_REQUESTED_VERSION 0.1 CACHE STRING "Required API version")
find_package(PRISM ${PRISM_REQUESTED_VERSION} CONFIG REQUIRED)
if(NOT PRISM_EXAMPLE_SOURCE)
    set(PRISM_EXAMPLE_SOURCE main.c)
endif()
add_executable(consumer "${PRISM_EXAMPLE_SOURCE}")
target_link_libraries(consumer PRIVATE prism::prism)
enable_testing()
add_test(NAME consumer COMMAND consumer)

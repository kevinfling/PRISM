cmake_minimum_required(VERSION 3.16)
project(PRISMSubprojectConsumer LANGUAGES C)
set(CMAKE_C_STANDARD 17)
add_subdirectory("${PRISM_SOURCE_DIR}" prism)
if(NOT CMAKE_C_STANDARD EQUAL 17)
    message(FATAL_ERROR "PRISM changed the parent C standard")
endif()
if(TARGET prism_example OR TARGET bench_prism OR TARGET test_core)
    message(FATAL_ERROR "PRISM enabled optional subproject targets by default")
endif()
add_executable(subproject_consumer ../consumer/main.c)
target_link_libraries(subproject_consumer PRIVATE prism::prism)
enable_testing()
add_test(NAME subproject_consumer COMMAND subproject_consumer)

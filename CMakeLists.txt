cmake_minimum_required(VERSION 3.16)
project(PRISM VERSION 0.1.0 LANGUAGES C)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
set(PRISM_STANDALONE OFF)
if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
    set(PRISM_STANDALONE ON)
endif()
option(PRISM_ENABLE_FFTW "Enable the FFTW transform backend" ON)
option(PRISM_BUILD_TESTS "Build correctness tests" ${PRISM_STANDALONE})
option(PRISM_BUILD_EXAMPLES "Build examples" ${PRISM_STANDALONE})
option(PRISM_BUILD_BENCHMARKS "Build benchmarks" ${PRISM_STANDALONE})
option(PRISM_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
option(PRISM_WARNINGS_AS_ERRORS "Treat PRISM compiler warnings as errors" OFF)
add_library(prism src/prism.c src/backend_reference.c src/backend_fftw.c)
add_library(prism::prism ALIAS prism)
target_compile_features(prism PUBLIC c_std_11)
set_target_properties(prism PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF
    VERSION ${PROJECT_VERSION} SOVERSION 0.1)
target_include_directories(prism PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
if(NOT MSVC)
    target_link_libraries(prism PUBLIC m)
endif()
if(PRISM_ENABLE_FFTW)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(FFTW3 REQUIRED IMPORTED_TARGET fftw3)
    target_link_libraries(prism PRIVATE PkgConfig::FFTW3)
    target_compile_definitions(prism PRIVATE PRISM_HAVE_FFTW=1)
endif()
function(prism_checks target)
    set_target_properties(${target} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
        if(PRISM_WARNINGS_AS_ERRORS)
            target_compile_options(${target} PRIVATE -Werror)
        endif()
        if(PRISM_SANITIZE)
            target_compile_options(${target} PRIVATE -g -fno-omit-frame-pointer -fsanitize=address,undefined)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        endif()
    endif()
endfunction()
if(PRISM_SANITIZE AND NOT CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    message(FATAL_ERROR "PRISM_SANITIZE requires GCC or Clang")
endif()
prism_checks(prism)
if(PRISM_SANITIZE AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_link_options(prism INTERFACE -fsanitize=address,undefined)
endif()
if(PRISM_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests)
endif()
if(PRISM_BUILD_EXAMPLES)
    add_subdirectory(examples)
endif()
if(PRISM_BUILD_BENCHMARKS)
    add_subdirectory(benchmarks)
endif()
install(TARGETS prism EXPORT PRISMTargets
    ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/PRISM)
set(PRISM_INSTALL_CMAKEDIR ${CMAKE_INSTALL_LIBDIR}/cmake/PRISM)
configure_package_config_file(cmake/PRISMConfig.cmake.in
    ${CMAKE_CURRENT_BINARY_DIR}/PRISMConfig.cmake
    INSTALL_DESTINATION ${PRISM_INSTALL_CMAKEDIR})
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/PRISMConfigVersion.cmake
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion)
install(EXPORT PRISMTargets NAMESPACE prism:: DESTINATION ${PRISM_INSTALL_CMAKEDIR})
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/PRISMConfig.cmake
    ${CMAKE_CURRENT_BINARY_DIR}/PRISMConfigVersion.cmake DESTINATION ${PRISM_INSTALL_CMAKEDIR})
